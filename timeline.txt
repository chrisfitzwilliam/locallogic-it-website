    <script>
        // Subtle Mouse Parallax for Stars
        document.addEventListener('mousemove', (e) => {
            const stars = document.querySelector('.star-field');
            const x = (e.clientX / window.innerWidth) * 20;
            const y = (e.clientY / window.innerHeight) * 20;
            stars.style.transform = `translate3d(-${x}px, -${y}px, 0)`;
        });

        // Dynamic SVG Path Adjustment (Simplified logic)
        function updatePath() {
            const nodes = document.querySelectorAll('.star-anchor');
            const svg = document.querySelector('.filament-svg');
            const paths = document.querySelectorAll('.filament-svg path');
            
            if(window.innerWidth < 768) return;

            let pathData = `M 500,${nodes[0].offsetTop + 40}`;
            
            for (let i = 1; i < nodes.length; i++) {
                const prev = nodes[i-1].offsetTop + 40;
                const curr = nodes[i].offsetTop + 40;
                const cp1y = prev + (curr - prev) / 2;
                const cp2y = prev + (curr - prev) / 2;
                
                // Create a weaving curve
                const cp1x = i % 2 === 0 ? 350 : 650;
                const cp2x = i % 2 === 0 ? 650 : 350;
                
                pathData += ` C ${cp1x},${cp1y} ${cp2x},${cp2y} 500,${curr}`;
            }

            paths.forEach(p => p.setAttribute('d', pathData));
        }

        window.addEventListener('resize', updatePath);
        window.addEventListener('load', updatePath);
    </script>
